ALTER PROCEDURE [dbo].[SPR_UPD_CERRAR_TICKETS_AUSENTES]
(
    @PI_FECHA_SINCRONIZACION DATETIME
)
AS
BEGIN

    SET NOCOUNT ON;

    UPDATE TBL_TTICKET
    SET EST_NID = 2
    WHERE EST_NID = 1
    AND (TIC_DULTIMA_SINCRONIZACION IS NULL
    OR TIC_DULTIMA_SINCRONIZACION < @PI_FECHA_SINCRONIZACION)

END
