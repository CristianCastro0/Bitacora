ALTER PROCEDURE [dbo].[SPR_VALIDAR_ROL_ETAPA]
    @PI_IDASIGNACION BIGINT
AS
BEGIN
    SET NOCOUNT ON;
 
    SELECT CASE WHEN EXISTS (
        SELECT 1
        FROM TBL_TASIGNACION A
        INNER JOIN TBL_TTICKET T ON T.TIC_NID = A.TIC_NID
        INNER JOIN TBL_RUSUARIO U ON U.USU_NID = A.USU_NID
        INNER JOIN TBL_RETAPAROL ER ON ER.ETA_NID = T.ETA_NID AND ER.ROL_NID = U.ROL_NID
        WHERE A.ASG_NID = @PI_IDASIGNACION
    ) THEN 1 ELSE 0 END AS TieneAcceso
END
