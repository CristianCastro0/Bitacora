ALTER PROCEDURE [dbo].[SPR_GET_HORAS_BITACORA]
    @PI_IDASIGNACION BIGINT
AS
BEGIN

	SET NOCOUNT ON;

    DECLARE @HOR_NID BIGINT;
    SELECT @HOR_NID = HOR_NID FROM TBL_TASIGNACION
    WHERE ASG_NID = @PI_IDASIGNACION;

	    SELECT ISNULL(SUM(b.BIT_NHORASTRABAJADAS),0) 
        FROM TBL_TBITACORA b
        INNER JOIN TBL_TASIGNACION a ON a.ASG_NID = b.ASG_NID
        WHERE a.HOR_NID = @HOR_NID

END
