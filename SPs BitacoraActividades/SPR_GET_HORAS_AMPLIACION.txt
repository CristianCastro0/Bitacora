ALTER PROCEDURE [dbo].[SPR_GET_HORAS_AMPLIACION]
    @PI_IDASIGNACION BIGINT
AS
BEGIN

	SET NOCOUNT ON;

		DECLARE @HOR_NID BIGINT;
		SELECT @HOR_NID = HOR_NID FROM TBL_TASIGNACION 
		WHERE ASG_NID = @PI_IDASIGNACION;
	    
		SELECT ISNULL(SUM(r.AMP_NHORAS_AMPLIADAS), 0)
		FROM TBL_RAMPLIACION r
		INNER JOIN TBL_TASIGNACION a ON a.ASG_NID = r.ASG_NID
		WHERE a.HOR_NID = @HOR_NID

END
