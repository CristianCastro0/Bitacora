ALTER PROCEDURE [dbo].[SPR_IU_USUARIO_LOCAL]
    @PI_DIRECTORIO VARCHAR(50),
    @PI_CORREO VARCHAR(50),
    @PI_NOMBRE1 VARCHAR(50),
    @PI_NOMBRE2 VARCHAR(50),
    @PI_APELLIDO1 VARCHAR(50),
    @PI_APELLIDO2 VARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;
 
    DECLARE @NOM_NID BIGINT;
 
    IF EXISTS (SELECT 1 FROM TBL_RUSUARIO WHERE USU_CDIRECTORIO = @PI_DIRECTORIO)
    BEGIN
        UPDATE TBL_RUSUARIO
        SET USU_CCORREO = @PI_CORREO
        WHERE USU_CDIRECTORIO = @PI_DIRECTORIO;
    END
    ELSE
    BEGIN
        INSERT INTO TBL_RNOMBRE (NOM_CNOMBRE1, NOM_CNOMBRE2, NOM_CAPELLIDO1, NOM_CAPELLIDO2)
        VALUES (@PI_NOMBRE1, @PI_NOMBRE2, @PI_APELLIDO1, @PI_APELLIDO2);
 
        SET @NOM_NID = SCOPE_IDENTITY();
 
        INSERT INTO TBL_RUSUARIO (NOM_NID, USU_CCORREO, USU_CDIRECTORIO, ROL_NID, ESU_NID, USU_CDOCUMENTO)
        VALUES (@NOM_NID, @PI_CORREO, @PI_DIRECTORIO, 1, 1, NULL);
    END
 
    SELECT
        U.USU_NID AS Id,
        U.USU_CCORREO AS Correo,
        U.USU_CDIRECTORIO AS Directorio,
        U.ROL_NID AS IdRol,
        U.ESU_NID AS IdEstado,
        N.NOM_CNOMBRE1 + ' ' + ISNULL(N.NOM_CNOMBRE2 + ' ', '') + N.NOM_CAPELLIDO1 + ' ' + ISNULL(N.NOM_CAPELLIDO2, '') AS NombreCompleto
    FROM TBL_RUSUARIO U
    JOIN TBL_RNOMBRE N ON N.NOM_NID = U.NOM_NID
    WHERE U.USU_CDIRECTORIO = @PI_DIRECTORIO;
END
