ALTER PROCEDURE [dbo].[SPR_GET_TICKETS_GRILLA]
AS
BEGIN

    SET NOCOUNT ON;

    SELECT
        T.TIC_NID AS Id,
        T.TIC_CCODIGO AS Codigo,
        T.TIC_CRESPONSABLE AS Senior,
        E.EST_CNOMBRE AS Estado,
        T.TIC_CDESCRIPCION AS Descripcion,
        T.TIC_DFECHA_CREACION AS FechaCreacion
    FROM TBL_TTICKET T
    INNER JOIN TBL_RESTADOTICKET E
        ON E.EST_NID = T.EST_NID
    ORDER BY T.TIC_DFECHA_CREACION DESC

END
