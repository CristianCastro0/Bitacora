ALTER PROCEDURE [dbo].[SPR_UPD_USUARIO]
    @PI_IDUSUARIO BIGINT,
    @PI_NOMBRE1 VARCHAR(30),
    @PI_NOMBRE2 VARCHAR(30),
    @PI_APELLIDO1 VARCHAR(30),
    @PI_APELLIDO2 VARCHAR(30),
    @PI_CORREO VARCHAR(30),
    @PI_ROLID BIGINT,
    @PI_ESTADOID BIGINT
AS
BEGIN

	    UPDATE TBL_RUSUARIO
            SET 
            USU_CCORREO = @PI_CORREO,
            ROL_NID = @PI_ROLID,
            ESU_NID = @PI_ESTADOID
            WHERE USU_NID = @PI_IDUSUARIO

        
        UPDATE n
            SET 
            NOM_CNOMBRE1 = @PI_NOMBRE1,
            NOM_CNOMBRE2 = @PI_NOMBRE2,
            NOM_CAPELLIDO1 = @PI_APELLIDO1,
            NOM_CAPELLIDO2 = @PI_APELLIDO2
            FROM TBL_RNOMBRE n
            INNER JOIN TBL_RUSUARIO u ON u.NOM_NID = n.NOM_NID
            WHERE u.USU_NID = @PI_IDUSUARIO

END
