ALTER PROCEDURE [dbo].[SPR_GET_TICKET]
    @PI_IDUSUARIO VARCHAR(50)
AS
BEGIN

	SET NOCOUNT ON;

	    SELECT t.TIC_NID AS Id, t.TIC_CCODIGO AS Codigo, t.TIC_CDESCRIPCION AS Nombre
        FROM TBL_TTICKET t
        INNER JOIN TBL_TASIGNACION a
        ON t.TIC_NID = a.TIC_NID
        INNER JOIN TBL_RUSUARIO u
        ON a.USU_NID = u.USU_NID
        WHERE u.USU_NID = @PI_IDUSUARIO
        AND t.EST_NID = 1 AND a.ESA_NID = 1
END
